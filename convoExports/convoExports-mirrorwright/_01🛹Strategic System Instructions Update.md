import Ajv from 'ajv';
import addFormats from 'ajv-formats';
import protocolSchema from './schemas/protocol.schema.json';

const ajv = new Ajv({ strict: true, allErrors: true });
addFormats(ajv);

const validate = ajv.compile(protocolSchema);

export function validateProtocol(data: any): { valid: boolean; errors?: string[] } {
  const isValid = validate(data);
  return {
    valid: isValid,
    errors: isValid ? undefined : ajv.errorsText(validate.errors, { separator: '\n' }).split('\n'),
  };
}
